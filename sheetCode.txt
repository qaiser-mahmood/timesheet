function doGet(e) {
  return doPost(e);
}

function doPost(e) {
  var JSON_STRING_REPLY;
    
  try {
    var requestData = {};
    if (e && e.postData && e.postData.contents) {
      try {
        requestData = JSON.parse(e.postData.contents);
      } catch (parseErr) {
        requestData = { action: "get" };
      }
    } else if (e && e.parameter && Object.keys(e.parameter).length > 0) {
      requestData = e.parameter;
    } else {
      requestData = { action: "get" };
    }
    var targetSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Roster");
    var staffSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Staff");

    // ACTION 1: FETCH DATA
    if (requestData.action === "get") {
      var logs = [];
      if (targetSheet) {
        var logRows = targetSheet.getDataRange().getDisplayValues();
        for (var i = 1; i < logRows.length; i++) {
          if (logRows[i][2]) {
            var cleanIsoDate = logRows[i][0];
            if (cleanIsoDate && cleanIsoDate.includes('/')) {
               try { cleanIsoDate = new Date(logRows[i][0]).toISOString().split('T')[0];
               } catch(err){}
            }
            logs.push({
              date: cleanIsoDate,
              weekCommencing: logRows[i][1],
              name: logRows[i][2].toString().trim(),
              from: logRows[i][3],
              to: logRows[i][4],
              cashRate: Number(logRows[i][5]) || 0,
              taxRate: Number(logRows[i][6]) || 0,
              totalHours: Number(logRows[i][7]) || 0,
              cashHours: Number(logRows[i][8]) || 0,
              taxHours: Number(logRows[i][9]) || 0
            });
          }
        }
      }
      
      var dynamicStaff = [];
      if (staffSheet) {
        var staffRows = staffSheet.getDataRange().getValues();
        for (var j = 1; j < staffRows.length; j++) {
          if (staffRows[j][0]) {
            var sName = staffRows[j][0].toString().trim();
            var rawCash = staffRows[j][1];
            var rawTax = staffRows[j][2];
            var isOwner = (sName.toLowerCase() === "qaiser");
            
            var cRate = (isOwner || rawCash === 0 || rawCash === "0") ? 0 : 
                        (rawCash !== "" && rawCash !== null && !isNaN(rawCash) ? Number(rawCash) : 25.00);
            var tRate = (isOwner || rawTax === 0 || rawTax === "0") ? 0 : 
                        (rawTax !== "" && rawTax !== null && !isNaN(rawTax) ? Number(rawTax) : 30.00);

            dynamicStaff.push({
              name: sName,
              cashRate: cRate,
              taxRate: tRate,
              status: staffRows[j][3] // Assuming column 4 is status
            });
          }
        }
      }
      var salesSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HourlySales");
      var salesData = [];
      if (salesSheet) {
        var salesRows = salesSheet.getDataRange().getDisplayValues();
        for (var s = 1; s < salesRows.length; s++) {
          if (salesRows[s][0]) {
            var cleanSalesDate = salesRows[s][0];
            if (cleanSalesDate.includes('/')) {
              try { cleanSalesDate = new Date(salesRows[s][0]).toISOString().split('T')[0]; } catch(e){}
            }
            var parsedHourly = {};
            try {
              parsedHourly = JSON.parse(salesRows[s][2]);
            } catch(e){}
            salesData.push({
              date: cleanSalesDate,
              totalSales: Number(salesRows[s][1]) || 0,
              hourly: parsedHourly,
              updatedAt: salesRows[s][3] || ""
            });
          }
        }
      }

      JSON_STRING_REPLY = JSON.stringify({ logs: logs, staff: dynamicStaff, sales: salesData });
    }
    
    // ACTION 2: SAVE ENTRY
    if (requestData.action === "save") {
      if (!targetSheet) {
        targetSheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Roster");
        targetSheet.appendRow(["Date", "Week Commencing", "Name", "From", "To", "Cash Rate", "Tax Rate", "Total Hours", "Cash Hours", "Tax Hours"]);
      }
      var entry = requestData.entry;
      var existingRows = targetSheet.getDataRange().getDisplayValues();
      var targetRowIndex = -1;
      var targetName = (entry.name || "").toString().trim().toLowerCase();
      for (var k = 1; k < existingRows.length; k++) {
        if (existingRows[k][0] === entry.date && existingRows[k][2].toString().trim().toLowerCase() === targetName) {
          targetRowIndex = k + 1;
          break;
        }
      }
      
      var payload = [entry.date, entry.weekCommencing, entry.name, entry.from, entry.to, entry.cashRate, entry.taxRate, entry.totalHours, entry.cashHours, entry.taxHours];
      if (targetRowIndex !== -1) {
        targetSheet.getRange(targetRowIndex, 1, 1, 10).setValues([payload]);
      } else {
        targetSheet.appendRow(payload);
      }
      if (targetSheet.getLastRow() > 1) {
        targetSheet.getRange(2, 1, targetSheet.getLastRow() - 1, 10).sort({ column: 1, ascending: true });
      }
      JSON_STRING_REPLY = JSON.stringify({ status: "success" });
    }
    
    // ACTION 3: DELETE ENTRY
    if (requestData.action === "delete") {
      if (targetSheet) {
        var currentData = targetSheet.getDataRange().getDisplayValues();
        var delName = (requestData.name || "").toString().trim().toLowerCase();
        for (var m = currentData.length - 1; m >= 1; m--) {
          if (currentData[m][0] === requestData.date && currentData[m][2].toString().trim().toLowerCase() === delName) {
            targetSheet.deleteRow(m + 1);
          }
        }
      }
      JSON_STRING_REPLY = JSON.stringify({ status: "deleted" });
    }


    // ACTION 4: COPY PREVIOUS WEEK ROSTER
    if (requestData.action === "copy_previous_week") {
      var targetWeekMonday = requestData.targetWeekMonday;
      var copiedCount = 0;
      
      if (targetSheet && targetWeekMonday) {
        var parts = targetWeekMonday.split('-').map(Number);
        var targetDate = new Date(Date.UTC(parts[0], parts[1] - 1, parts[2]));
        targetDate.setUTCDate(targetDate.getUTCDate() - 7);
        var prevWeekMonday = targetDate.toISOString().split('T')[0];
        
        var existingRows = targetSheet.getDataRange().getDisplayValues();
        var prevWeekShifts = [];
        
        for (var i = 1; i < existingRows.length; i++) {
          var row = existingRows[i];
          if (row[2] && row[0]) {
            var cleanDate = row[0];
            if (cleanDate.includes('/')) {
              try { cleanDate = new Date(row[0]).toISOString().split('T')[0]; } catch(e){}
            }
            var rowWeekCommencing = row[1];
            if (rowWeekCommencing && rowWeekCommencing.includes('/')) {
              try { rowWeekCommencing = new Date(row[1]).toISOString().split('T')[0]; } catch(e){}
            }
            if (rowWeekCommencing === prevWeekMonday) {
              prevWeekShifts.push({
                date: cleanDate,
                name: row[2].toString().trim(),
                from: row[3],
                to: row[4],
                cashRate: Number(row[5]) || 0,
                taxRate: Number(row[6]) || 0,
                totalHours: Number(row[7]) || 0,
                cashHours: Number(row[8]) || 0,
                taxHours: Number(row[9]) || 0
              });
            }
          }
        }
        
        for (var p = 0; p < prevWeekShifts.length; p++) {
          var s = prevWeekShifts[p];
          var sParts = s.date.split('-').map(Number);
          var sDate = new Date(Date.UTC(sParts[0], sParts[1] - 1, sParts[2]));
          sDate.setUTCDate(sDate.getUTCDate() + 7);
          var newShiftDate = sDate.toISOString().split('T')[0];
          
          var newPayload = [newShiftDate, targetWeekMonday, s.name, s.from, s.to, s.cashRate, s.taxRate, s.totalHours, s.cashHours, s.taxHours];
          
          var targetRowIndex = -1;
          var currentData = targetSheet.getDataRange().getDisplayValues();
          for (var k = 1; k < currentData.length; k++) {
            if (currentData[k][0] === newShiftDate && currentData[k][2].toString().trim() === s.name) {
              targetRowIndex = k + 1;
              break;
            }
          }
          
          if (targetRowIndex !== -1) {
            targetSheet.getRange(targetRowIndex, 1, 1, 10).setValues([newPayload]);
          } else {
            targetSheet.appendRow(newPayload);
          }
          copiedCount++;
        }
      }
      if (targetSheet.getLastRow() > 1) {
        targetSheet.getRange(2, 1, targetSheet.getLastRow() - 1, 10).sort({ column: 1, ascending: true });
      }
      JSON_STRING_REPLY = JSON.stringify({ status: "success", count: copiedCount });
    }

    // ACTION 5: SAVE HOURLY SALES (Single or Multi-Day Batch)
    if (requestData.action === "save_hourly_sales" || requestData.action === "save_multiple_hourly_sales") {
      var salesSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HourlySales");
      if (!salesSheet) {
        salesSheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("HourlySales");
        salesSheet.appendRow(["Date", "Total Sales", "Hourly JSON", "Last Updated"]);
      }
      var sUpdated = new Date().toISOString();
      var existingSalesRows = salesSheet.getDataRange().getDisplayValues();
      var dateToRow = {};
      for (var r = 1; r < existingSalesRows.length; r++) {
        if (existingSalesRows[r][0]) {
          dateToRow[existingSalesRows[r][0]] = r + 1;
        }
      }

      var daysToSave = [];
      if (requestData.days && Array.isArray(requestData.days)) {
        daysToSave = requestData.days;
      } else if (requestData.date) {
        daysToSave.push({
          date: requestData.date,
          totalSales: requestData.totalSales,
          hourly: requestData.hourly
        });
      }

      for (var d = 0; d < daysToSave.length; d++) {
        var dayItem = daysToSave[d];
        var sDate = dayItem.date;
        var sTotal = Number(dayItem.totalSales) || 0;
        var sHourlyJson = typeof dayItem.hourly === "string" ? dayItem.hourly : JSON.stringify(dayItem.hourly || {});
        var salesPayload = [sDate, sTotal, sHourlyJson, sUpdated];

        if (dateToRow[sDate]) {
          salesSheet.getRange(dateToRow[sDate], 1, 1, 4).setValues([salesPayload]);
        } else {
          salesSheet.appendRow(salesPayload);
          dateToRow[sDate] = salesSheet.getLastRow();
        }
      }
      if (salesSheet.getLastRow() > 1) {
        salesSheet.getRange(2, 1, salesSheet.getLastRow() - 1, 4).sort({ column: 1, ascending: true });
      }
      JSON_STRING_REPLY = JSON.stringify({ status: "success", count: daysToSave.length });
    }

    // ACTION 6: SAVE RAW TRANSACTIONS LOG (De-duplicated & Ascending Date Order)
    if (requestData.action === "save_raw_transactions" && requestData.transactions) {
      var rawSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("RawTransactions");
      if (!rawSheet) {
        rawSheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("RawTransactions");
        rawSheet.appendRow(["Date", "Time", "Amount", "Reference / Raw Line", "Logged At"]);
      }
      var existingRaw = rawSheet.getDataRange().getDisplayValues();
      var existingKeys = {};
      for (var r = 1; r < existingRaw.length; r++) {
        var eDate = existingRaw[r][0];
        var eTime = existingRaw[r][1];
        var eAmt = parseFloat(existingRaw[r][2]) || 0;
        var eRaw = existingRaw[r][3] || "";
        var key = (eDate + "|" + eTime + "|" + eAmt.toFixed(2) + "|" + eRaw).toLowerCase();
        existingKeys[key] = true;
      }

      var nowIso = new Date().toISOString();
      var rawItems = requestData.transactions;
      var rawPayloads = [];
      for (var ti = 0; ti < rawItems.length; ti++) {
        var it = rawItems[ti];
        var itAmt = parseFloat(it.amount) || 0;
        var itKey = (it.date + "|" + it.time + "|" + itAmt.toFixed(2) + "|" + (it.raw || "")).toLowerCase();
        if (!existingKeys[itKey]) {
          existingKeys[itKey] = true;
          rawPayloads.push([it.date, it.time, itAmt, it.raw || "", nowIso]);
        }
      }
      if (rawPayloads.length > 0) {
        rawSheet.getRange(rawSheet.getLastRow() + 1, 1, rawPayloads.length, 5).setValues(rawPayloads);
      }
      if (rawSheet.getLastRow() > 1) {
        rawSheet.getRange(2, 1, rawSheet.getLastRow() - 1, 5).sort([
          { column: 1, ascending: true },
          { column: 2, ascending: true }
        ]);
      }
      JSON_STRING_REPLY = JSON.stringify({ status: "success", count: rawPayloads.length });
    }

    
  } catch (error) {
    JSON_STRING_REPLY = JSON.stringify({ status: "error", message: error.toString() });
  }
  return ContentService.createTextOutput(JSON_STRING_REPLY).setMimeType(ContentService.MimeType.JSON);
}
